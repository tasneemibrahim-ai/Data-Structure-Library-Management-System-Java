// Product Class
class Product {
    int productId;
    String productName;
    double price;
    int quantityInStock;

    // Constructor for creating a new product
    public Product(int productId, String productName, double price, int quantityInStock) {
        this.productId = productId;
        this.productName = productName;
        this.price = price;
        this.quantityInStock = quantityInStock;
    }

    // Getter methods for accessing product attributes
    public int getProductId() {
        return productId;
    }

    public String getProductName() {
        return productName;
    }

    public double getPrice() {
        return price;
    }

    public int getQuantityInStock() {
        return quantityInStock;
    }

    // Setter methods for updating product attributes
    public void setPrice(double price) {
        this.price = price;
    }

    public void setQuantityInStock(int quantityInStock) {
        this.quantityInStock = quantityInStock;
    }

    // toString method for displaying product information
    @Override
    public String toString() {
        return "Product ID: " + productId +
                "\nProduct Name: " + productName +
                "\nPrice: $" + price +
                "\nQuantity in Stock: " + quantityInStock;
    }
}

    // Inventory Class
class Inventory {
    private List<Product> products;

    public Inventory() {
        this.products = new ArrayList<>();
    }

    // Add a new product to the inventory
    public void addProduct(Product product) {
        products.add(product);
        // Assume sorting based on productId for simplicity
        Collections.sort(products, Comparator.comparingInt(p -> p.productId));
        System.out.println("Product " + product.productName + " added to the inventory.");
    }

    // Update stock levels for a specific product
    public void updateStock(int productId, int quantity) {
        for (Product product : products) {
            if (product.productId == productId) {
                product.quantityInStock += quantity;
                System.out.println("Stock for product " + product.productName + " updated to " + product.quantityInStock);
                break;
            }
        }
    }

    // Retrieve product information by productId
    public Product getProductById(int productId) {
        for (Product product : products) {
            if (product.productId == productId) {
                return product;
            }
        }
        System.out.println("Product with ID " + productId + " not found in the inventory.");
        return null;
    }

    // Get a sorted list of products based on product ID
    public List<Product> getSortedProductList() {
        // Assume sorting based on productId for simplicity
        List<Product> sortedProducts = new ArrayList<>(products);
        Collections.sort(sortedProducts, Comparator.comparingInt(p -> p.productId));
        return sortedProducts;
    }
}


// Order Class
class Order {
    int orderId;
    String customerName;
    List<Product> productsOrdered;
    double totalAmount;

    public Order(int orderId, String customerName) {
        this.orderId = orderId;
        this.customerName = customerName;
        this.productsOrdered = new ArrayList<>();
        this.totalAmount = 0.0;
    }

    // Add a product to the order
    public void addProduct(Product product) {
        productsOrdered.add(product);
        calculateTotalAmount();
        System.out.println("Product " + product.productName + " added to Order #" + orderId);
    }

    // Calculate the total amount for the order
    private void calculateTotalAmount() {
        totalAmount = 0.0;
        for (Product product : productsOrdered) {
            totalAmount += product.price;
        }
    }

    // Get the total amount for the order
    public double getTotalAmount() {
        return totalAmount;
    }

    // Get a list of products in the order
    public List<Product> getProductsOrdered() {
        return productsOrdered;
    }
}
// OrderQueue Class
class OrderQueue {
    private Queue<Order> orders;

    public OrderQueue() {
        this.orders = new LinkedList<>();
    }

    // Enqueue a new order to the queue
    public void enqueueOrder(Order order) {
        orders.add(order);
        System.out.println("Order #" + order.orderId + " added to the queue.");
    }

    // Dequeue the next processed order from the queue
    public Order dequeueOrder() {
        if (!orders.isEmpty()) {
            Order processedOrder = orders.poll();
            System.out.println("Order #" + processedOrder.orderId + " processed and removed from the queue.");
            return processedOrder;
        } else {
            System.out.println("No orders in the queue.");
            return null;
        }
    }

    // Peek at the next order in the queue without removing it
    public Order peekNextOrder() {
        if (!orders.isEmpty()) {
            return orders.peek();
        } else {
            System.out.println("No orders in the queue.");
            return null;
        }
    }

    // Get the size of the queue
    public int getSize() {
        return orders.size();
    }
}

// Customer Interface
public class CustomerInterface {

    private Inventory inventory;
    private List<Order> orders;

    public CustomerInterface(Inventory inventory) {
        this.inventory = inventory;
        this.orders = new ArrayList<>();
    }

    // Display available products to customers
    public void displayAvailableProducts() {
        List<Product> products = inventory.getSortedProductList();
        System.out.println("Available Products:");
        for (Product product : products) {
            System.out.println(product.getProductId() + ". " + product.getProductName() +
                    " - $" + product.getPrice() + " (Stock: " + product.getQuantityInStock() + ")");
        }
    }

    // Add items to the customer's cart
    public void addToCart(int productId, int quantity) {
        Product product = inventory.getProductById(productId);
        if (product != null && product.getQuantityInStock() >= quantity) {
            Order currentOrder = getCurrentOrder();
            if (currentOrder == null) {
                currentOrder = new Order(orders.size() + 1, "Customer");
                orders.add(currentOrder);
            }
            currentOrder.addProduct(product, quantity);
            System.out.println(quantity + " " + product.getProductName() + "(s) added to your cart.");
        } else {
            System.out.println("Invalid product or insufficient stock.");
        }
    }

    // Place an order and provide feedback on order status and estimated delivery time
    public void placeOrder() {
        Order currentOrder = getCurrentOrder();
        if (currentOrder != null && !currentOrder.getProductsOrdered().isEmpty()) {
            double totalAmount = currentOrder.getTotalAmount();
            System.out.println("Order placed successfully! Your total amount is: $" + totalAmount);
            System.out.println("Estimated delivery time: 3 days");
        } else {
            System.out.println("No items in your cart. Please add items before placing an order.");
        }
    }

    // Get the current order for the customer
    private Order getCurrentOrder() {
        if (!orders.isEmpty()) {
            return orders.get(orders.size() - 1);
        }
        return null;
    }
}

// Inventory Class
class Inventory {
    private List<Product> products;

    public Inventory() {
        this.products = new ArrayList<>();
    }

    // Add a new product to the inventory
    public void addProduct(Product product) {
        products.add(product);
        // Maintain the product list in sorted order based on product ID
        mergeSort(products, Comparator.comparingInt(Product::getProductId));
    }

    // Merge Sort implementation
    private <T> void mergeSort(List<T> list, Comparator<? super T> comparator) {
        if (list.size() > 1) {
            int mid = list.size() / 2;
            List<T> left = new ArrayList<>(list.subList(0, mid));
            List<T> right = new ArrayList<>(list.subList(mid, list.size()));

            mergeSort(left, comparator);
            mergeSort(right, comparator);

            merge(list, left, right, comparator);
        }
    }

    private <T> void merge(List<T> list, List<T> left, List<T> right, Comparator<? super T> comparator) {
        int i = 0, j = 0, k = 0;

        while (i < left.size() && j < right.size()) {
            if (comparator.compare(left.get(i), right.get(j)) <= 0) {
                list.set(k++, left.get(i++));
            } else {
                list.set(k++, right.get(j++));
            }
        }

        while (i < left.size()) {
            list.set(k++, left.get(i++));
        }

        while (j < right.size()) {
            list.set(k++, right.get(j++));
        }
    }

    // Display the sorted product list
    public void displaySortedProductList() {
        for (Product product : products) {
            System.out.println("Product ID: " + product.getProductId() +
                    ", Product Name: " + product.getProductName() +
                    ", Price: $" + product.getPrice() +
                    ", Quantity in Stock: " + product.getQuantityInStock());
        }
    }
}
